## Technical Implementation

### Step 1: Create API Client Hook
**File:** `AB0-1-front/lib/hooks/useCompanyAnalytics.ts`
```typescript
import { useQuery } from '@tanstack/react-query';
import { companyDashboardApi } from '@/lib/api/company-dashboard';

export interface AnalyticsOverview {
  views_30d: number;
  cta_clicks_30d: number;
  whatsapp_clicks_30d: number;
  leads_30d: number;
  conversion_rate: number;
  data_source: string;
  freshness: {
    last_updated_at: string;
  };
}

export function useCompanyAnalytics(companyId: number) {
  return useQuery({
    queryKey: ['company-analytics-overview', companyId],
    queryFn: () => companyDashboardApi.getAnalyticsOverview(companyId),
    refetchInterval: 30000, // 30 seconds auto-refresh
    staleTime: 60000, // 1 minute cache
    retry: 3,
    retryDelay: (attemptIndex) => Math.min(1000 * 2 ** attemptIndex, 10000),
  });
}
```

### Step 2: Create API Method
**File:** `AB0-1-front/lib/api/company-dashboard.ts`
```typescript
import type { AnalyticsOverview } from '@/lib/hooks/useCompanyAnalytics';

export const companyDashboardApi = {
  async getAnalyticsOverview(companyId: number): Promise<AnalyticsOverview> {
    const response = await fetch(
      `/api/v1/company_dashboard/analytics/overview?company_id=${companyId}`,
      {
        headers: {
          'Authorization': `Bearer ${getAuthToken()}`,
          'Content-Type': 'application/json',
        },
      }
    );

    if (!response.ok) {
      throw new Error(`Failed to fetch analytics: ${response.statusText}`);
    }

    return response.json();
  },
};
```

### Step 3: Update PerformanceMetrics Component
**File:** `AB0-1-front/app/dashboard/components/PerformanceMetrics.tsx`
```typescript
import { useCompanyAnalytics } from '@/lib/hooks/useCompanyAnalytics';
import { Skeleton } from '@/components/ui/skeleton';
import { AlertCircle } from 'lucide-react';

interface PerformanceMetricsProps {
  companyId: number;
}

export function PerformanceMetrics({ companyId }: PerformanceMetricsProps) {
  const { data, isLoading, isError, error } = useCompanyAnalytics(companyId);

  // Loading State
  if (isLoading) {
    return (
      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
        {Array.from({ length: 4 }).map((_, i) => (
          <div key={i} className="p-4 border rounded-lg">
            <Skeleton className="h-4 w-20 mb-2" />
            <Skeleton className="h-8 w-16" />
          </div>
        ))}
      </div>
    );
  }

  // Error State
  if (isError) {
    return (
      <div className="p-4 border border-red-200 rounded-lg bg-red-50">
        <div className="flex items-center gap-2 text-red-800">
          <AlertCircle className="w-5 h-5" />
          <span className="font-medium">Erro ao carregar métricas</span>
        </div>
        <p className="text-sm text-red-600 mt-1">
          {error instanceof Error ? error.message : 'Erro desconhecido'}
        </p>
      </div>
    );
  }

  // Success State - Real Data
  return (
    <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
      <MetricCard
        label="Views (30d)"
        value={data.views_30d}
        icon="👁️"
      />
      <MetricCard
        label="CTA Clicks"
        value={data.cta_clicks_30d}
        icon="👆"
      />
      <MetricCard
        label="WhatsApp Clicks"
        value={data.whatsapp_clicks_30d}
        icon="💬"
      />
      <MetricCard
        label="Leads"
        value={data.leads_30d}
        icon="📊"
      />
      <MetricCard
        label="Conversão"
        value={`${data.conversion_rate.toFixed(1)}%`}
        icon="🎯"
        className="col-span-2 md:col-span-4"
      />
      
      {/* Data Freshness Indicator */}
      <div className="col-span-2 md:col-span-4 text-xs text-muted-foreground text-right">
        Atualizado: {new Date(data.freshness.last_updated_at).toLocaleString('pt-BR')}
      </div>
    </div>
  );
}
```

### Step 4: Verify Backend Endpoint
**File:** `AB0-1-back/app/controllers/api/v1/company_dashboard_controller.rb`
```ruby
# Ensure endpoint returns correct format
def analytics_overview
  metrics = CompanyDashboard::MetricsSource.new(@company).call
  
  render json: {
    views_30d: metrics[:views_30d],
    cta_clicks_30d: metrics[:cta_clicks_30d],
    whatsapp_clicks_30d: metrics[:whatsapp_clicks_30d],
    leads_30d: metrics[:leads_30d],
    conversion_rate: calculate_conversion_rate(metrics),
    data_source: 'company_daily_stats',
    freshness: {
      last_updated_at: Time.current.iso8601
    }
  }
end

private

def calculate_conversion_rate(metrics)
  return 0 if metrics[:views_30d].zero?
  (metrics[:leads_30d].to_f / metrics[:views_30d] * 100).round(2)
end
```

import { renderHook, waitFor } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';

const queryClient = new QueryClient();

    global.fetch = jest.fn();

  it('fetches analytics data successfully', async () => {
    (global.fetch as jest.Mock).mockResolvedValue({

        views_30d: 1234,
        cta_clicks_30d: 567,
        whatsapp_clicks_30d: 89,
        leads_30d: 45,
        conversion_rate: 3.65
      })
    });

    const { result } = renderHook(() => useCompanyAnalytics(123), { wrapper });

    expect(result.current.data?.views_30d).toBe(1234);

## Data Validation Query
Run this SQL to verify backend data matches frontend display:
```sql
SELECT 
  c.id AS company_id,
  c.name,
  SUM(cds.profile_views) AS views_30d,
  SUM(cds.cta_clicks) AS cta_clicks_30d,
  SUM(cds.whatsapp_clicks) AS whatsapp_clicks_30d,
  SUM(cds.leads) AS leads_30d,
  ROUND(
    (SUM(cds.leads)::decimal / NULLIF(SUM(cds.profile_views), 0)) * 100, 
    2
  ) AS conversion_rate
FROM companies c
LEFT JOIN company_daily_stats cds 
  ON cds.company_id = c.id 
  AND cds.date >= CURRENT_DATE - INTERVAL '30 days'
WHERE c.id = :company_id
GROUP BY c.id, c.name;
```
