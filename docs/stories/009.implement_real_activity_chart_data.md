```typescript
async function track(eventType: string, properties: Record<string, any>) {
  try {
    await fetch('/api/v1/analytics/track', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${getAuthToken()}`
      },
      body: JSON.stringify({
        event_type: eventType,
        properties
      })
    });
  } catch (error) {
    console.error('Analytics tracking error:', error);
  }
}
```

### Step 2: Create Analytics Service (Backend)
**File:** `AB0-1-back/app/services/review_dashboard/activity_service.rb`
```ruby
module ReviewDashboard
  class ActivityService
    def initialize(user)
      @user = user
      @start_date = 30.days.ago.to_date
      @end_date = Date.current
    end

    def call
      {
        activity_30d: generate_activity_chart
      }
    end

    private

    def generate_activity_chart
      date_range = (@start_date..@end_date).to_a

      # Fetch all events in one query
      events_by_date = fetch_events_grouped_by_date

      # Fetch leads (CTA clicks)
      leads_by_date = fetch_leads_grouped_by_date

      # Build chart data
      date_range.map do |date|
        {
          date: date.to_s,
          profile_views: events_by_date.dig(date, 'profile_view') || 0,
          whatsapp_clicks: events_by_date.dig(date, 'whatsapp_click') || 0,
          cta_clicks: leads_by_date[date] || 0
        }
      end
    end

    def fetch_events_grouped_by_date
      AnalyticsEvent
        .where(user_id: @user.id)
        .where(event_type: %w[profile_view whatsapp_click])
        .where('DATE(created_at) >= ? AND DATE(created_at) <= ?', @start_date, @end_date)
        .group('DATE(created_at)', 'event_type')
        .count
        .transform_keys { |key| [key[0], key[1]] } # [date, event_type]
        .each_with_object({}) do |((date, event_type), count), hash|
          hash[date] ||= {}
          hash[date][event_type] = count
        end
    end

    def fetch_leads_grouped_by_date
      Lead
        .where(email: @user.email)
        .where('DATE(created_at) >= ? AND DATE(created_at) <= ?', @start_date, @end_date)
        .group('DATE(created_at)')
        .count
    end
  end
end
```

### Step 4: Add Analytics Endpoint Authorization
**File:** `AB0-1-back/app/controllers/api/v1/analytics_controller.rb`
```ruby
class Api::V1::AnalyticsController < ApplicationController
  before_action :authenticate_api_user, only: [:track]

  def track
    event_type = params[:event_type]
    properties = params[:properties] || {}

    # Server-side validation
    unless valid_event_type?(event_type)
      render json: { error: 'Invalid event type' }, status: :unprocessable_entity
      return
    end

    # Create event with user context
    AnalyticsEvent.create!(
      event_type: event_type,
      user_id: current_user.id,
      company_id: properties[:company_id],
      properties: properties,
      ip_address: request.remote_ip,
      user_agent: request.user_agent
    )

    head :no_content
  rescue StandardError => e
    Rails.logger.error("Analytics tracking error: #{e.message}")
    head :no_content # Don't block user experience
  end

  private

  def valid_event_type?(type)
    %w[
      profile_view
      whatsapp_click
      cta_click
      review_created
      lead_created
    ].include?(type)
  end
end
```

```ruby
require 'rails_helper'

RSpec.describe ReviewDashboard::ActivityService do
  let(:user) { create(:user) }
  let(:service) { described_class.new(user) }

  describe '#call' do
    before do
      # Create events for last 30 days
      create(:analytics_event, 
        user: user, 
        event_type: 'profile_view', 
        created_at: 2.days.ago
      )
      create(:analytics_event, 
        user: user, 
        event_type: 'profile_view', 
        created_at: 2.days.ago
      )
      create(:analytics_event, 
        user: user, 
        event_type: 'whatsapp_click', 
        created_at: 5.days.ago
      )
      create(:lead, email: user.email, created_at: 3.days.ago)
    end

    it 'returns activity data for 30 days' do
      result = service.call

      expect(result[:activity_30d]).to be_an(Array)
      expect(result[:activity_30d].length).to eq(31) # 30 days + today
    end

    it 'includes real event counts' do
      result = service.call
      two_days_ago_data = result[:activity_30d].find { |d| d[:date] == 2.days.ago.to_date.to_s }

      expect(two_days_ago_data[:profile_views]).to eq(2)
    end

    it 'includes whatsapp clicks' do
      result = service.call
      five_days_ago_data = result[:activity_30d].find { |d| d[:date] == 5.days.ago.to_date.to_s }

      expect(five_days_ago_data[:whatsapp_clicks]).to eq(1)
    end

    it 'includes CTA clicks from leads' do
      result = service.call
      three_days_ago_data = result[:activity_30d].find { |d| d[:date] == 3.days.ago.to_date.to_s }

      expect(three_days_ago_data[:cta_clicks]).to eq(1)
    end

    it 'shows 0 for days with no activity' do
      result = service.call
      yesterday_data = result[:activity_30d].find { |d| d[:date] == 1.day.ago.to_date.to_s }

      expect(yesterday_data[:profile_views]).to eq(0)
      expect(yesterday_data[:whatsapp_clicks]).to eq(0)
      expect(yesterday_data[:cta_clicks]).to eq(0)
    end
  end
end
```

```ruby
require 'rails_helper'

RSpec.describe 'Api::V1::Analytics', type: :request do
  let(:user) { create(:user) }

  before { sign_in user }

  describe 'POST /api/v1/analytics/track' do
    it 'tracks profile_view event' do
      expect {
        post '/api/v1/analytics/track',
             params: {
               event_type: 'profile_view',
               properties: { company_id: 123 }
             },
             headers: auth_headers(user)
      }.to change(AnalyticsEvent, :count).by(1)

      expect(response).to have_http_status(:no_content)

      event = AnalyticsEvent.last
      expect(event.event_type).to eq('profile_view')
      expect(event.user_id).to eq(user.id)
      expect(event.properties['company_id']).to eq(123)
    end

    it 'rejects invalid event types' do
      post '/api/v1/analytics/track',
           params: { event_type: 'invalid_event' },
           headers: auth_headers(user)

      expect(response).to have_http_status(:unprocessable_entity)
    end

    it 'requires authentication' do
      post '/api/v1/analytics/track',
           params: { event_type: 'profile_view' }

      expect(response).to have_http_status(:unauthorized)
    end
  end
end
```
