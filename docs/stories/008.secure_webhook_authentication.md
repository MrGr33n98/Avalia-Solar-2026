## Context & Problem

```ruby

# payments_webhooks_controller.rb:5

# RISCO: Qualquer payload pode simular pagamentos
# IMPACTO: Fraude em transações
```

**Severity:** High - Financial fraud risk  
**Attack Vector:** Unauthenticated webhook endpoint accepts any POST payload  
**Business Impact:** Potential financial loss, compromised transaction integrity

### Step 2: Create Webhook Security Service
**File:** `AB0-1-back/app/services/webhook_security_service.rb`
```ruby
class WebhookSecurityService
  class InvalidSignatureError < StandardError; end
  class TimestampError < StandardError; end

  def initialize(provider:, payload:, signature:, timestamp:)
    @provider = provider
    @payload = payload
    @signature = signature
    @timestamp = timestamp
    @secret = Rails.application.config.webhook.secret
    @tolerance = Rails.application.config.webhook.signature_tolerance
  end

  def verify!
    verify_timestamp!
    verify_signature!
    true
  end

  private

  def verify_timestamp!
    return if @timestamp.blank? # Allow for providers without timestamp

    timestamp_time = Time.at(@timestamp.to_i)
    age = (Time.current - timestamp_time).abs

    if age > @tolerance
      raise TimestampError, "Timestamp too old: #{age}s (max: #{@tolerance}s)"
    end
  end

  def verify_signature!
    expected_signature = generate_signature

    unless ActiveSupport::SecurityUtils.secure_compare(@signature, expected_signature)
      raise InvalidSignatureError, "Invalid signature for provider: #{@provider}"
    end
  end

  def generate_signature
    case @provider
    when 'stripe'
      generate_stripe_signature
    when 'mock'
      generate_hmac_signature
    else
      raise ArgumentError, "Unknown provider: #{@provider}"
    end
  end

  def generate_stripe_signature
    # Stripe format: t=timestamp,v1=signature
    signed_payload = "#{@timestamp}.#{@payload}"
    OpenSSL::HMAC.hexdigest('SHA256', @secret, signed_payload)
  end

  def generate_hmac_signature
    # Simple HMAC for mock provider
    OpenSSL::HMAC.hexdigest('SHA256', @secret, @payload)
  end
end
```

### Step 3: Update Webhook Controller
**File:** `AB0-1-back/app/controllers/api/v1/payments_webhooks_controller.rb`
```ruby
class Api::V1::PaymentsWebhooksController < ApplicationController
  skip_before_action :verify_authenticity_token
  before_action :verify_webhook_signature

  def create
    checkout_session_id = params[:checkout_session_id]
    
    session = PaymentCheckoutSession.find_by(checkout_session_id: checkout_session_id)
    
    unless session
      render json: { error: 'Session not found' }, status: :not_found
      return
    end

    # Process webhook event
    process_payment_event(session, params[:event_type])

    render json: { success: true }, status: :ok
  rescue StandardError => e
    Rails.logger.error("Webhook processing error: #{e.message}")
    render json: { error: e.message }, status: :unprocessable_entity
  end

  private

  def verify_webhook_signature
    provider = params[:provider]
    signature = request.headers['X-Webhook-Signature']
    timestamp = request.headers['X-Webhook-Timestamp']
    payload = request.body.read

    # Reset body for subsequent reads
    request.body.rewind

    WebhookSecurityService.new(
      provider: provider,
      payload: payload,
      signature: signature,
      timestamp: timestamp
    ).verify!

  rescue WebhookSecurityService::InvalidSignatureError => e
    log_security_event('invalid_signature', provider, signature)
    render json: { error: 'Invalid signature' }, status: :unauthorized
  rescue WebhookSecurityService::TimestampError => e
    log_security_event('timestamp_expired', provider, timestamp)
    render json: { error: 'Timestamp expired' }, status: :unauthorized
  end

  def log_security_event(event_type, provider, details)
    Rails.logger.warn(
      "Webhook security event: #{event_type}, " \
      "provider: #{provider}, details: #{details}, " \
      "ip: #{request.remote_ip}"
    )

    # Optional: Store in database for audit
    WebhookSecurityLog.create!(
      event_type: event_type,
      provider: provider,
      details: details,
      ip_address: request.remote_ip,
      created_at: Time.current
    )
  end

  def process_payment_event(session, event_type)
    case event_type
    when 'checkout.session.completed'
      session.mark_as_paid!
      notify_company(session.company)
    when 'checkout.session.expired'
      session.mark_as_expired!
    else
      Rails.logger.info("Unhandled event type: #{event_type}")
    end
  end
end
```

## Testing Strategy

### Unit Tests - Service
**File:** `AB0-1-back/spec/services/webhook_security_service_spec.rb`
```ruby
require 'rails_helper'

RSpec.describe WebhookSecurityService do
  let(:secret) { 'test-webhook-secret-123456789012' }
  let(:payload) { '{"event":"test"}' }
  let(:timestamp) { Time.current.to_i.to_s }

  before do
    allow(Rails.application.config.webhook).to receive(:secret).and_return(secret)
    allow(Rails.application.config.webhook).to receive(:signature_tolerance).and_return(300)
  end

  describe '#verify!' do
    context 'with valid HMAC signature' do
      let(:signature) { OpenSSL::HMAC.hexdigest('SHA256', secret, payload) }

      it 'returns true' do
        service = described_class.new(
          provider: 'mock',
          payload: payload,
          signature: signature,
          timestamp: timestamp
        )

        expect(service.verify!).to be true
      end
    end

    context 'with invalid signature' do
      let(:signature) { 'invalid-signature-12345' }

      it 'raises InvalidSignatureError' do
        service = described_class.new(
          provider: 'mock',
          payload: payload,
          signature: signature,
          timestamp: timestamp
        )

        expect { service.verify! }.to raise_error(
          WebhookSecurityService::InvalidSignatureError
        )
      end
    end

    context 'with expired timestamp' do
      let(:old_timestamp) { (Time.current - 10.minutes).to_i.to_s }
      let(:signature) { OpenSSL::HMAC.hexdigest('SHA256', secret, payload) }

      it 'raises TimestampError' do
        service = described_class.new(
          provider: 'mock',
          payload: payload,
          signature: signature,
          timestamp: old_timestamp
        )

        expect { service.verify! }.to raise_error(
          WebhookSecurityService::TimestampError
        )
      end
    end

    context 'with Stripe provider' do
      let(:signed_payload) { "#{timestamp}.#{payload}" }
      let(:signature) { OpenSSL::HMAC.hexdigest('SHA256', secret, signed_payload) }

      it 'verifies Stripe signature format' do
        service = described_class.new(
          provider: 'stripe',
          payload: payload,
          signature: signature,
          timestamp: timestamp
        )

        expect(service.verify!).to be true
      end
    end
  end
end
```

```bash
#!/bin/bash

SECRET="test-webhook-secret-123456789012"
PAYLOAD='{"checkout_session_id":"session_123","event_type":"checkout.session.completed"}'
TIMESTAMP=$(date +%s)

# Generate valid signature
SIGNATURE=$(echo -n "$PAYLOAD" | openssl dgst -sha256 -hmac "$SECRET" | cut -d' ' -f2)

# Test valid webhook
echo "Testing VALID webhook..."
curl -X POST http://localhost:3000/api/v1/payments/webhooks/mock \
  -H "Content-Type: application/json" \
  -H "X-Webhook-Signature: $SIGNATURE" \
  -H "X-Webhook-Timestamp: $TIMESTAMP" \
  -d "$PAYLOAD"

echo -e "\n\nTesting INVALID signature..."
curl -X POST http://localhost:3000/api/v1/payments/webhooks/mock \
  -H "Content-Type: application/json" \
  -H "X-Webhook-Signature: invalid-signature-123" \
  -H "X-Webhook-Timestamp: $TIMESTAMP" \
  -d "$PAYLOAD"

echo -e "\n\nTesting EXPIRED timestamp (11 minutes ago)..."
OLD_TIMESTAMP=$((TIMESTAMP - 660))
```
