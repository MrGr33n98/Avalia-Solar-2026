## Technical Implementation

### Step 1: Create Type Guard Function
**File:** `AB0-1-front/lib/utils/company-helpers.ts`
```typescript
interface CompanyInfo {
  name: string;
  initials: string;
  logoUrl: string | null;
}

export function extractCompanyInfo(
  company: string | { name: string; logo_url?: string } | undefined
): CompanyInfo {
  // Handle undefined
  if (!company) {
    return {
      name: 'Empresa',
      initials: '??',
      logoUrl: null
    };
  }

  // Handle object format (current backend)
  if (typeof company === 'object') {
    return {
      name: company.name,
      initials: company.name.substring(0, 2).toUpperCase(),
      logoUrl: company.logo_url || null
    };
  }

  // Handle legacy string format (backward compatibility)
  return {
    name: company,
    initials: company.substring(0, 2).toUpperCase(),
    logoUrl: null
  };
}
```

### Step 4: Add Unit Tests
**File:** `AB0-1-front/__tests__/utils/company-helpers.test.ts`

```typescript
import { extractCompanyInfo } from '@/lib/utils/company-helpers';

describe('extractCompanyInfo', () => {
  it('handles object format', () => {
    const result = extractCompanyInfo({
      name: 'Solar Tech',
      logo_url: 'https://example.com/logo.png'
    });
    
    expect(result.name).toBe('Solar Tech');
    expect(result.initials).toBe('SO');
    expect(result.logoUrl).toBe('https://example.com/logo.png');
  });

  it('handles string format (legacy)', () => {
    const result = extractCompanyInfo('Solar Tech');
    
    expect(result.name).toBe('Solar Tech');
    expect(result.initials).toBe('SO');
    expect(result.logoUrl).toBeNull();
  });

  it('handles undefined', () => {
    const result = extractCompanyInfo(undefined);
    
    expect(result.name).toBe('Empresa');
    expect(result.initials).toBe('??');
    expect(result.logoUrl).toBeNull();
  });

  it('handles object without logo', () => {
    const result = extractCompanyInfo({
      name: 'ABC Company'
    });
    
    expect(result.logoUrl).toBeNull();
  });
});
```
